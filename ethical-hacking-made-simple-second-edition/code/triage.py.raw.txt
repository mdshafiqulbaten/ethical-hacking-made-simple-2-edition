      counts.values())
      print("sha256 :", hashlib.sha256(data).hexdigest())
      print("size   :", len(data), "bytes")
      print(f"entropy: {entropy:.2f} bits per byte")
      for s in re.findall(rb"[ -~]{6,}", data)[:10]:
